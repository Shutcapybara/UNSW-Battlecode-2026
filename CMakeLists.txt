cmake_minimum_required(VERSION 3.16)
project(BattlecodeBots LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(danger_levels bots/danger-levels/main.cpp)
target_compile_options(danger_levels PRIVATE -Wall -Wextra -Wpedantic)
add_executable(dragon_hunter bots/dragon-hunters/hunter.cpp)
target_compile_options(dragon_hunter PRIVATE -Wall -Wextra -Wpedantic)
add_executable(escorts bots/escorts/escort.cpp)
target_compile_options(escorts PRIVATE -Wall -Wextra -Wpedantic)
add_executable(kamikaze_swarm bots/kamikaze-swarm/swarm.cpp)
target_compile_options(kamikaze_swarm PRIVATE -Wall -Wextra -Wpedantic)
add_executable(one_child bots/one-child/one_child.cpp)
target_compile_options(one_child PRIVATE -Wall -Wextra -Wpedantic)
add_executable(pearl_seeker bots/pearl-seeker/main.cpp)
target_compile_options(pearl_seeker PRIVATE -Wall -Wextra -Wpedantic)
add_executable(pearl_seeker_center bots/pearl-seeker-center/main.cpp)
target_compile_options(pearl_seeker_center PRIVATE -Wall -Wextra -Wpedantic)
add_executable(pre_swarm_defense bots/pre-swarm-defense/baseline.cpp)
target_compile_options(pre_swarm_defense PRIVATE -Wall -Wextra -Wpedantic)
add_executable(two_children bots/two-children/two_children.cpp)
target_compile_options(two_children PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME bot_behaviour COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_bot.py $<TARGET_FILE:danger_levels>)
    add_test(NAME pearl_seeker_behaviour COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_pearl_seeker.py $<TARGET_FILE:pearl_seeker>)
    add_test(NAME dragon_hunter_behaviour COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_hunter.py $<TARGET_FILE:dragon_hunter>)
    add_test(NAME escort_behaviour COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_escorts.py $<TARGET_FILE:escorts>)
endif()
